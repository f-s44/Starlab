from machine import Pin, SPI
import time
# ---------------- PIN DEFINITIONS ----------------
SPI_PORT = 0       # Corresponds to spi1
SPI_SCK = 18       # GP10 is hardware SPI1 SCK
SPI_MOSI = 19      # GP11 is hardware SPI1 TX (MOSI)
SPI_MISO = 16      # GP12 is hardware SPI1 RX (MISO)
SPI_CS = 17
SPI_BAUDRATE = 1_000_000  # 1 MHz

# ---------------- ACS REGISTERS ----------------
REG_POWER_DATA = 0x20
txBuffer = bytearray(8)
# ---------------- SPI SETUP ----------------
spi = SPI(SPI_PORT,
          baudrate=SPI_BAUDRATE,
          polarity=1,
          phase=1,
          sck=Pin(SPI_SCK),
          mosi=Pin(SPI_MOSI),
          miso=Pin(SPI_MISO))
cs = Pin(SPI_CS, Pin.OUT)
cs.value(1)  # CS high initially

# ---------------- SPI READ ----------------
def read_register(reg):
    cmd = bytearray(4)
    cmd[0] = reg # command byte with R/W bit = 0 for read
    cmd[1] = 0
    cmd[2] = 0 
    cmd[3] = 0
    response = bytearray(4)
    cs.value(0)
    spi.write_readinto(cmd, response) 
    cs.value(1)
    # Convert 4 bytes into a 32-bit integer
    val = (response[0] << 24) | (response[1] << 16) | (response[2] << 8) | response[3]
    return val
# ---------------- PACKET BUILD ----------------
def build_packet(voltage, current):
    v_fixed = int(voltage * 100) # convert to fixed-point (2 decimal places)
    i_fixed = int(current * 100) # convert to fixed-point (2 decimal places)
    txBuffer[0] = 0xAA # start byte
    txBuffer[1] = (v_fixed >> 8) & 0xFF # voltage high byte
    txBuffer[2] = v_fixed & 0xFF # voltage low byte
    txBuffer[3] = (i_fixed >> 8) & 0xFF # current high byte
    txBuffer[4] = i_fixed & 0xFF # current low byte
    txBuffer[5] = 0x00 # reserved byte
    txBuffer[6] = (txBuffer[1] + txBuffer[2] + txBuffer[3] + txBuffer[4]) & 0xFF 
    #  checksum
    txBuffer[7] = 0x55 # end byte

# ---------------- MAIN LOOP ----------------
print("Pico 2 Power Monitor Ready")
while True:
    packed_data = read_register(REG_POWER_DATA)

    vrms_raw = packed_data & 0xFFFF # lower 16 bits for voltage
    irms_raw = (packed_data >> 16) & 0xFFFF # upper 16 bits for current


    voltage = (vrms_raw / 65536.0) * 0.84 * 1.19
    current = (irms_raw / 65536.0) * 15.0 * 1.19

    build_packet(voltage, current)

    print("RAW:", vrms_raw, irms_raw)
    #print("Voltage: {:.2f} V | Current: {:.2f} A".format(voltage, current))
    time.sleep_ms(200)
