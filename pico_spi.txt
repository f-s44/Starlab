from machine import Pin, mem32
import time

# ---------------- PIN DEFINITIONS ----------------
SPI_SCK = 18
SPI_MOSI = 19
SPI_MISO = 16
SPI_CS = 17

REG_POWER_DATA = 0x20

txBuffer = bytearray(8)

# ---------------- SPI0 REGISTER MAP ----------------
SPI0_BASE = 0x4003C000

SSPCR0  = SPI0_BASE + 0x000
SSPCR1  = SPI0_BASE + 0x004
SSPDR   = SPI0_BASE + 0x008
SSPSR   = SPI0_BASE + 0x00C
SSPCPSR = SPI0_BASE + 0x010

# Status flags
SPI_SR_TNF = 1 << 1
SPI_SR_RNE = 1 << 2
SPI_SR_BSY = 1 << 4

# ---------------- PIN SETUP ----------------
cs = Pin(SPI_CS, Pin.OUT)
cs.value(1)

Pin(SPI_SCK, Pin.ALT)
Pin(SPI_MOSI, Pin.ALT)
Pin(SPI_MISO, Pin.ALT)

# ---------------- SPI INIT ----------------
mem32[SSPCR1] = 0        # disable SPI

mem32[SSPCPSR] = 6       # clock prescaler

mem32[SSPCR0] = (
        (7 << 0) |       # 8-bit data
        (1 << 6) |       # CPOL = 1
        (1 << 7)         # CPHA = 1
)

mem32[SSPCR1] = 2        # enable SPI

# ---------------- LOW LEVEL TRANSFER ----------------
def spi_transfer(tx):

    rx = bytearray(len(tx))

    for i in range(len(tx)):

        while not (mem32[SSPSR] & SPI_SR_TNF):
            pass

        mem32[SSPDR] = tx[i]

        while not (mem32[SSPSR] & SPI_SR_RNE):
            pass

        rx[i] = mem32[SSPDR] & 0xFF

    return rx

# ---------------- SPI READ ----------------
def read_register(reg):

    addr = (reg & 0x7F) << 1
    read_cmd = addr | 1

    tx = bytearray(5)
    tx[0] = read_cmd

    cs.value(0)
    spi_transfer(tx)
    cs.value(1)

    time.sleep_us(5)

    cs.value(0)
    rx = spi_transfer(tx)
    cs.value(1)

    val = (rx[1] << 24) | (rx[2] << 16) | (rx[3] << 8) | rx[4]

    return val

# ---------------- MAIN LOOP ----------------
print("Pico 2 Direct-SPI Power Monitor Ready")

while True:

    packed_data = read_register(REG_POWER_DATA)

    vrms_raw = packed_data & 0xFFFF
    irms_raw = (packed_data >> 16) & 0xFFFF

    if irms_raw & 0x8000:
        irms_raw -= 0x10000

    voltage = vrms_raw * 0.84 * 1.19
    current = irms_raw * 15.0 * 1.19

    #print("RAW:", vrms_raw, irms_raw)
    print("Voltage: {:.2f} V, Current: {:.2f} A".format(voltage, current))
    time.sleep_ms(200)
