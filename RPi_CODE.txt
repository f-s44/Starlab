from smbus2 import SMBus
import time

bus = SMBus(1)
PICO1 = 0x10
PICO2 = 0x11

def read_v_i(addr):
    data = bus.read_i2c_block_data(addr, 0, 4)
    voltage = int.from_bytes(data[:2], 'big') / 100
    current = int.from_bytes(data[2:], 'big') / 100
    return voltage, current

while True:
    v1, i1 = read_v_i(PICO1)
    v2, i2 = read_v_i(PICO2)
    print(f"Pico1 → Voltage: {v1:.2f} V, Current: {i1:.2f} A")
    print(f"Pico2 → Voltage: {v2:.2f} V, Current: {i2:.2f} A")
    time.sleep(1)
