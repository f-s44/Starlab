
#include <SPI.h>
#include <mcp_can.h>

#define CS_CAN 5
#define CS_ACS 10

MCP_CAN CAN(CS_CAN);
SPIClass spi = SPIClass(FSPI);

uint32_t readACS(uint8_t addr) {
  digitalWrite(CS_ACS, LOW);
  spi.transfer(addr);
  uint32_t value = 0;
  value |= spi.transfer(0x00) << 24;
  value |= spi.transfer(0x00) << 16;
  value |= spi.transfer(0x00) << 8;
  value |= spi.transfer(0x00);
  digitalWrite(CS_ACS, HIGH);
  return value;
}

void setup() {
  Serial.begin(115200);

  spi.begin(12, 13, 11, CS_ACS);
  pinMode(CS_ACS, OUTPUT);
  digitalWrite(CS_ACS, HIGH);

  if (CAN.begin(MCP_ANY, 500000, MCP_16MHZ) == CAN_OK)
    Serial.println("CAN init OK");
  else
    Serial.println("CAN init FAIL");

  CAN.setMode(MCP_NORMAL);
}

void loop() {

  uint32_t voltage_raw = readACS(0x2A);
  uint32_t current_raw = readACS(0x2C);

  uint16_t voltage = voltage_raw >> 16;
  uint16_t current = current_raw >> 16;

  byte data[4];
  data[0] = voltage >> 8;
  data[1] = voltage & 0xFF;
  data[2] = current >> 8;
  data[3] = current & 0xFF;

  CAN.sendMsgBuf(0x100, 0, 4, data);

  delay(1000);
}
